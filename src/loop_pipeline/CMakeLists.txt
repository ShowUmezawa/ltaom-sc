cmake_minimum_required(VERSION 3.10)
project(loop_pipeline)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
find_package(catkin REQUIRED COMPONENTS roscpp rospy std_msgs std_srvs pcl_ros cv_bridge)
find_package(PCL REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Ceres REQUIRED)
find_package(Threads REQUIRED)
catkin_package()
set(SC_SOURCE_DIR "" CACHE PATH "Pinned upstream Scan Context cpp/module/Scancontext directory")
if(NOT EXISTS "${SC_SOURCE_DIR}/Scancontext.cpp")
  message(FATAL_ERROR "Set SC_SOURCE_DIR to the pinned source in dependency_lock.json")
endif()
include_directories(${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${CERES_INCLUDE_DIRS}
  ${SC_SOURCE_DIR} ../STD)
add_executable(sc_candidates src/sc_candidates.cpp ${SC_SOURCE_DIR}/Scancontext.cpp)
target_link_libraries(sc_candidates ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${OpenCV_LIBS})
add_executable(std_candidates src/std_candidates.cpp ../STD/src/std.cpp)
target_link_libraries(std_candidates ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${OpenCV_LIBS} ${CERES_LIBRARIES} tbb)
catkin_install_python(PROGRAMS scripts/pipeline.py scripts/verify.py scripts/live_pipeline.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
set(D2_DEPS_DIR "" CACHE PATH "Pinned Quatro/Nano-GICP/TEASER++ dependency sources")
if(D2_DEPS_DIR)
  include(registration_deps.cmake)
endif()
