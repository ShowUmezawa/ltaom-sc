cmake_minimum_required(VERSION 3.16)
project(ltaom_frontend)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs visualization_msgs pcl_conversions tf2 tf2_ros ltaom_interfaces ltaom_transport)
 find_package(${dep} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS common io filters kdtree registration)
find_package(Eigen3 REQUIRED)
set(FROZEN_EIGEN_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/eigen3" CACHE PATH "Frozen frontend Eigen 3.4.0")
set(EXPECTED_EIGEN_MAJOR 4)
set(EXPECTED_EIGEN_MINOR 0)
if(NOT EXISTS "${FROZEN_EIGEN_INCLUDE}/Eigen/Core")
 message(FATAL_ERROR "Explicit frozen Eigen and version required")
endif()
include_directories(BEFORE ${FROZEN_EIGEN_INCLUDE})
add_compile_definitions(LTAOM_EXPECTED_EIGEN_MAJOR=${EXPECTED_EIGEN_MAJOR} LTAOM_EXPECTED_EIGEN_MINOR=${EXPECTED_EIGEN_MINOR})
find_package(OpenCV REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system)
find_package(Python3 REQUIRED COMPONENTS Development)
add_compile_definitions(ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/" LTAOM_STANDALONE MP_EN MP_PROC_NUM=4)
include_directories(include src ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${Python3_INCLUDE_DIRS})
add_library(extra_lib SHARED src/extra_lib.cpp)
ament_target_dependencies(extra_lib rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs visualization_msgs pcl_conversions tf2 tf2_ros ltaom_interfaces ltaom_transport)
target_link_libraries(extra_lib Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS} OpenMP::OpenMP_CXX)
add_executable(fastlio_node src/laserMapping.cpp src/preprocess.cpp include/ikd-Tree/ikd_Tree.cpp include/ikd-Forest/ikd_Forest.cpp include/FOV_Checker/FOV_Checker.cpp)
ament_target_dependencies(fastlio_node rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs visualization_msgs pcl_conversions tf2 tf2_ros ltaom_interfaces ltaom_transport)
target_link_libraries(fastlio_node extra_lib Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS} Boost::filesystem Boost::system Python3::Python OpenMP::OpenMP_CXX)
install(TARGETS fastlio_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS extra_lib LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
ament_export_libraries(extra_lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 add_executable(imu_bias_regression test/imu_bias_regression.cpp)
 target_compile_definitions(imu_bias_regression PRIVATE LTAOM_TEST_ROS2)
 ament_target_dependencies(imu_bias_regression rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs visualization_msgs pcl_conversions tf2 tf2_ros ltaom_interfaces ltaom_transport)
 target_link_libraries(imu_bias_regression Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS} OpenMP::OpenMP_CXX)
 add_test(NAME imu_bias_prediction_and_deskew COMMAND imu_bias_regression)
endif()
ament_package()
