cmake_minimum_required(VERSION 3.16)
project(ltaom_loop_core)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
include_directories(${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/Scancontext)
add_executable(sc_candidates src/sc_candidates.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/Scancontext/Scancontext.cpp)
target_link_libraries(sc_candidates ${PCL_LIBRARIES} ${OpenCV_LIBS})
set(D2_DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/registration")
include(registration_deps.cmake)
install(TARGETS sc_candidates registration DESTINATION lib/${PROJECT_NAME})
ament_package()
