cmake_minimum_required(VERSION 3.16)
project(ltaom_std)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ltaom_transport REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Ceres REQUIRED)
find_package(OpenSSL REQUIRED)
include_directories(. include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${CERES_INCLUDE_DIRS})
add_library(std_core SHARED src/std.cpp src/std_ba.cpp)
ament_target_dependencies(std_core ltaom_transport pcl_conversions)
target_link_libraries(std_core ${PCL_LIBRARIES} ${OpenCV_LIBS} ${CERES_LIBRARIES} tbb)
add_executable(std_node src/loop_detection.cpp)
ament_target_dependencies(std_node ltaom_transport pcl_conversions)
target_link_libraries(std_node std_core OpenSSL::Crypto)
install(TARGETS std_core LIBRARY DESTINATION lib)
install(TARGETS std_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_libraries(std_core)
ament_package()
