cmake_minimum_required(VERSION 3.16)
project(ltaom_backend)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(GTSAM REQUIRED)
find_package(ltaom_transport REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(OpenSSL REQUIRED)
include_directories(include ${GTSAM_INCLUDE_DIR} ${PCL_INCLUDE_DIRS})
add_executable(rebuild_factor_archive src/rebuild_factor_archive.cpp)
target_link_libraries(rebuild_factor_archive gtsam)
add_executable(isam_snapshot_test src/isam_snapshot_test.cpp)
target_link_libraries(isam_snapshot_test gtsam)
add_executable(backend_node src/backend_node.cpp)
ament_target_dependencies(backend_node ltaom_transport pcl_conversions)
target_link_libraries(backend_node gtsam ${PCL_LIBRARIES} OpenSSL::Crypto)
install(TARGETS backend_node rebuild_factor_archive isam_snapshot_test DESTINATION lib/${PROJECT_NAME})
ament_package()
